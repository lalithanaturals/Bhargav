function doGet(e) { const action = e.parameter.action; const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (action === 'search') { const query = (e.parameter.query || '').toString().trim().toLowerCase(); const callback = e.parameter.callback || 'handleSearchCallback'; const rows = sheet.getDataRange().getValues(); let data = []; if (rows.length > 1 && query !== '') { const headers = rows[0]; for (let i = 1; i < rows.length; i++) { let obj = { rowIndex: i + 1 }; let match = false; for (let j = 0; j < headers.length; j++) { let headerName = headers[j] ? headers[j].toString().trim() : 'col_' + j; let val = rows[i][j]; obj[headerName] = val; if (String(val).toLowerCase().includes(query)) { match = true; } } if (match) { // Fallback calculations in case fields are empty/NaN in old rows if (obj.alAmount === "" || obj.alAmount === undefined || isNaN(obj.alAmount)) { obj.alAmount = (parseFloat(obj.alNet || obj.alGross || 0) * 150); } if (obj.stAmount === "" || obj.stAmount === undefined || isNaN(obj.stAmount)) { obj.stAmount = (parseFloat(obj.stNet || obj.stGross || 0) * 50); } if (obj.grandTotal === "" || obj.grandTotal === undefined || isNaN(obj.grandTotal)) { obj.grandTotal = parseFloat(obj.alAmount) + parseFloat(obj.stAmount); } data.push(obj); } } data.reverse(); } return ContentService.createTextOutput(callback + '(' + JSON.stringify(data) + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } // Default action: Generate Next Exchange & Estimate IDs const rows = sheet.getDataRange().getValues(); let maxExNum = 0; let maxEstNum = 0; for (let i = 1; i < rows.length; i++) { let row = rows[i]; for (let col = 0; col < row.length; col++) { let val = String(row[col] || ''); if (val.startsWith('LNex')) { let num = parseInt(val.replace('LNex', '')) || 0; if (num > maxExNum) maxExNum = num; } else if (val.startsWith('Est')) { let num = parseInt(val.replace('Est', '')) || 0; if (num > maxEstNum) maxEstNum = num; } } } const nextId = 'LNex' + (maxExNum + 1); const nextEstId = 'Est' + (maxEstNum + 1); const responseObj = { nextId: nextId, nextEstId: nextEstId }; const idCallback = e.parameter.callback || 'handleIdCallback'; return ContentService.createTextOutput(idCallback + '(' + JSON.stringify(responseObj) + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } function doPost(e) { const lock = LockService.getScriptLock(); lock.tryLock(10000); try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const params = e.parameter; if (sheet.getLastRow() === 0) { sheet.appendRow([ 'timestamp', 'exchangeId', 'estId', 'customerName', 'customerPhone', 'alGross', 'alNet', 'alAmount', 'stGross', 'stNet', 'stAmount', 'grandTotal', 'type' ]); } const timestamp = params.timestamp || new Date().toLocaleString(); const exchangeId = params.exchangeId || ''; const estId = params.estId || ''; const customerName = params.customerName || ''; const customerPhone = params.customerPhone || ''; const alGross = params.alGross || 0; const alNet = params.alNet || 0; const alAmount = params.alAmount || 0; const stGross = params.stGross || 0; const stNet = params.stNet || 0; const stAmount = params.stAmount || 0; const grandTotal = params.grandTotal || 0; const type = params.type || 'TRANSACTION'; if (params.convertFromRow) { let rowIdx = parseInt(params.convertFromRow); if (rowIdx > 1) { sheet.getRange(rowIdx, 2).setValue(exchangeId); } } sheet.appendRow([ timestamp, exchangeId, estId, customerName, customerPhone, alGross, alNet, alAmount, stGross, stNet, stAmount, grandTotal, type ]); return ContentService.createTextOutput(JSON.stringify({ status: 'success' })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
